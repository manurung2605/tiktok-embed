        // Helper: Parse URL input entries
        function parseUrlsFromInput() {
            if (!urlInput || !urlInput.value) return [];
            const rawText = urlInput.value;
            if (!rawText.trim()) return [];
            
            const regex = /(?:https?:)?\/\/[^\s,]+|vt\.tiktok\.com\/[a-zA-Z0-9_-]+/gi;
            const matches = rawText.match(regex) || [];
            
            return matches.map(url => {
                let formatted = url.trim();
                if (formatted.startsWith('//')) {
                    formatted = 'https:' + formatted;
                } else if (!formatted.startsWith('http://') && !formatted.startsWith('https://')) {
                    formatted = 'https://' + formatted;
                }
                return formatted;
            });
        }

        function updateUrlCount() {
            if (!urlCounter) return;
            const urls = parseUrlsFromInput();
            urlCounter.textContent = `${urls.length} URL${urls.length === 1 ? '' : 's'} detected`;
        }

                type === 'error' ? 'bg-rose-950/90 border-rose-800 text-rose-200' :
                'bg-slate-900/90 border-slate-700 text-slate-200'
            }`;

                         type === 'error' ? 'fa-triangle-exclamation text-rose-400' :
                         'fa-circle-info text-cyan-400';

            toast.innerHTML = `<i class="fa-solid ${icon} text-base"></i><span>${message}</span>`;
            container.appendChild(toast);

            requestAnimationFrame(() => {
                toast.classList.remove('translate-y-2', 'opacity-0');
            });

            setTimeout(() => {
                toast.classList.add('opacity-0', 'translate-y-2');
                setTimeout(() => toast.remove(), 300);
            }, 3000);
        }

        function extractTikTokUrlFromText(text) {
            if (!text) return null;

            const videoPattern = /https?:\/\/(?:www\.)?tiktok\.com\/@[a-zA-Z0-9_.-]+\/video\/\d+/i;
            const match = text.match(videoPattern);
            if (match) return match[0];

            const photoPattern = /https?:\/\/(?:www\.)?tiktok\.com\/@[a-zA-Z0-9_.-]+\/photo\/\d+/i;
            const photoMatch = text.match(photoPattern);
            if (photoMatch) return photoMatch[0];

            return null;
        }

        // Server-based Unshortener Services Pipeline
        async function resolveSingleUrl(shortUrl) {
            const shortCodeMatch = shortUrl.match(/vt\.tiktok\.com\/([a-zA-Z0-9_-]+)/i);
            const shortCode = shortCodeMatch ? shortCodeMatch[1] : null;

            const serverServices = [
                // Server API 1: Direct fetch via unshorten.me endpoint
                async (url) => {
                    const response = await fetch(`https://unshorten.me/json/${encodeURIComponent(url)}`, { signal: AbortSignal.timeout(5000) });
                    if (response.ok) {
                        const data = await response.json();
                        if (data && data.resolved_url && data.resolved_url.includes('tiktok.com')) {
                            return data.resolved_url;
                        }
                    }
                    return null;
                },
                // Server API 2: AllOrigins relay for TikTok oEmbed API
                async (url) => {
                    const target = `https://www.tiktok.com/oembed?url=${encodeURIComponent(url)}`;
                    const response = await fetch(`https://api.allorigins.win/get?url=${encodeURIComponent(target)}`, { signal: AbortSignal.timeout(5000) });
                    if (response.ok) {
                        const data = await response.json();
                        if (data && data.contents) {
                            const parsed = extractTikTokUrlFromText(data.contents);
                            if (parsed) return parsed;
                        }
                    }
                    return null;
                },
                // Server API 3: CORS Proxy relay query
                async (url) => {
                    const target = `https://www.tiktok.com/oembed?url=${encodeURIComponent(url)}`;
                    const response = await fetch(`https://corsproxy.io/?${encodeURIComponent(target)}`, { signal: AbortSignal.timeout(5000) });
                    if (response.ok) {
                        const text = await response.text();
                        const parsed = extractTikTokUrlFromText(text);
                        if (parsed) return parsed;
                    }
                    return null;
                }
            ];

            for (const service of serverServices) {
                try {
                    const result = await service(shortUrl);
                    if (result && result.includes('tiktok.com')) {
                        return { success: true, url: result };
                    }
                } catch (e) {
                    // Try next resolver strategy
                }
            }

            // Fallback: If network APIs are restricted, build canonical path using shortcode
            if (shortCode) {
                return { success: true, url: `https://www.tiktok.com/t/${shortCode}/` };
            }

            return { success: false, url: shortUrl };
        }

        // Clean tracking query parameters
        function cleanTikTokUrl(url) {
            try {
                const parsed = new URL(url);
                parsed.search = '';
                return parsed.toString();
            } catch (e) {
                return url;
            }
        }

        // Batch Processing Engine
        async function startBatchResolution() {
            const urls = parseUrlsFromInput();
            if (urls.length === 0) {
                showToast('Please enter at least one valid TikTok short link!', 'error');
                return;
            }

            isProcessing = true;
            shouldStop = false;
            resolvedResults = [];
            
            startBtn.disabled = true;
            startBtn.classList.add('opacity-50', 'cursor-not-allowed');
            stopBtn.disabled = false;
            stopBtn.classList.remove('hidden');
            stopBtn.innerHTML = `<i class="fa-solid fa-hand"></i> Stop`;

            progressSection.classList.remove('hidden');
            resultsTableBody.innerHTML = '';
            resultBadge.textContent = `0 / ${urls.length}`;

            let completed = 0;
            const total = urls.length;
            const concurrency = 2;

            updateProgress(0, total);

            const queue = [...urls];
            const workers = Array(concurrency).fill(0).map(async () => {
                while (queue.length > 0 && !shouldStop) {
                    const shortUrl = queue.shift();
                    const index = urls.indexOf(shortUrl) + 1;

                    const rowId = `row-${index}`;
                    appendPlaceholderRow(rowId, index, shortUrl);

                    const res = await resolveSingleUrl(shortUrl);
                    
                    let finalUrl = res.url;
                    if (res.success && cleanParamsToggle.checked) {
                        finalUrl = cleanTikTokUrl(finalUrl);
                    }

                    const resultObj = {
                        id: index,
                        original: shortUrl,
                        full: finalUrl,
                        status: res.success ? 'Resolved' : 'Fallback Target',
                        isSuccess: res.success
                    };

                    resolvedResults.push(resultObj);
                    updateTableRow(rowId, resultObj);

                    completed++;
                    updateProgress(completed, total);
                    resultBadge.textContent = `${completed} / ${total}`;
                }
            });

            await Promise.all(workers);

            if (resolvedResults.length > 0) {
                generateHtmlEmbeds();
            }

            isProcessing = false;
            startBtn.disabled = false;
            startBtn.classList.remove('opacity-50', 'cursor-not-allowed');
            stopBtn.disabled = true;
            stopBtn.classList.add('hidden');

            if (shouldStop) {
                progressStatusText.innerHTML = `<i class="fa-solid fa-circle-pause text-amber-400"></i> Stopped by user`;
                showToast('Batch processing stopped.', 'info');
            } else {
                progressStatusText.innerHTML = `<i class="fa-solid fa-circle-check text-emerald-400"></i> Completed all items!`;
                showToast(`Resolved ${completed} links & generated embeds!`, 'success');
            }
        }

            progressBarInner.style.width = `${percent}%`;
            progressPercentText.textContent = `${percent}%`;
            progressCountText.textContent = `Resolved: ${completed} / ${total}`;
        }

        function copyAllResolved() {
            if (resolvedResults.length === 0) {
                showToast('No results available to copy!', 'error');
                return;
            }
            const urlsText = resolvedResults.map(r => r.full).join('\n');
            fallbackCopyText(urlsText, `Copied ${resolvedResults.length} resolved links!`);
        }

        function fallbackCopyText(text, successMsg) {
            if (navigator.clipboard && window.isSecureContext) {
                navigator.clipboard.writeText(text)
                    .then(() => showToast(successMsg, 'success'))
                    .catch(() => executeExecCommandCopy(text, successMsg));
            } else {
                executeExecCommandCopy(text, successMsg);
            }
        }

        function executeExecCommandCopy(text, successMessage) {
            const textArea = document.createElement("textarea");
            textArea.value = text;
            textArea.style.position = "fixed";
            document.body.appendChild(textArea);
            textArea.focus();
            textArea.select();
            try {
                document.execCommand('copy');
                showToast(successMessage, 'success');
            } catch (err) {
                showToast('Failed to copy text', 'error');
            }
            document.body.removeChild(textArea);
        }

        // Helper: Video ID Extractor
        function extractVideoId(url) {
            const match = url.match(/\/(?:video|photo)\/(\d+)/i);
            return match ? match[1] : '';
        }

        // Helper: Short Code Extractor
        function extractShortCode(url) {
            const match = url.match(/\/t\/([a-zA-Z0-9_-]+)/i) || url.match(/vt\.tiktok\.com\/([a-zA-Z0-9_-]+)/i);
            return match ? match[1] : '';
        }
